/* 사용자 정보 인터페이스용 테이블 생성 */
CREATE TABLE "BAT_UP_IMUSER" 
(
	"INS_TIME" VARCHAR2(20) NOT NULL ENABLE, 
	"EMP_NO" VARCHAR2(20), 
	"KOR_NM" VARCHAR2(100), 
	"DEPT_CD" VARCHAR2(20), 
	"POS_CD" VARCHAR2(20), 
	"EMAIL" VARCHAR2(100), 
	"HP_NO" VARCHAR2(100), 
	"STATUS" VARCHAR2(20), 
	"JOB_DATE" DATE, 
	"IF_FLAG" VARCHAR2(1), 
	 PRIMARY KEY ("INS_TIME", "EMP_NO")
);

COMMENT ON TABLE BAT_UP_IMUSER IS '[남부발전] 사용자 정보 인터페이스용 테이블';
COMMENT ON COLUMN BAT_UP_IMUSER.INS_TIME IS '생성일시';
COMMENT ON COLUMN BAT_UP_IMUSER.EMP_NO IS '사용자ID';
COMMENT ON COLUMN BAT_UP_IMUSER.KOR_NM IS '사용자명';
COMMENT ON COLUMN BAT_UP_IMUSER.DEPT_CD IS '부서코드';
COMMENT ON COLUMN BAT_UP_IMUSER.POS_CD IS '직책코드';
COMMENT ON COLUMN BAT_UP_IMUSER.EMAIL IS 'EMAIL 주소';
COMMENT ON COLUMN BAT_UP_IMUSER.HP_NO IS '전화번호';
COMMENT ON COLUMN BAT_UP_IMUSER.STATUS IS '재직여부';
COMMENT ON COLUMN BAT_UP_IMUSER.JOB_DATE IS '데이터 수신일시';
COMMENT ON COLUMN BAT_UP_IMUSER.IF_FLAG IS '처리여부';


/* 부서 정보 인터페이스용 테이블 생성 */

CREATE TABLE "BAT_UP_IMDEPT" 
(	"INS_TIME" TIMESTAMP NOT NULL, 
	"DEPT_CODE" VARCHAR2(20) NOT NULL, 
	"DEPT_NAME" VARCHAR2(100),
	"UP_DEPT_CD" VARCHAR2(20), 
	"PRNT_CD" VARCHAR2(20), 
	"MANAGER_ID" VARCHAR2(20), 
	"IF_FLAG" VARCHAR2(1),

COMMENT ON TABLE BAT_UP_IMDEPT IS '[남부발전] 부서 정보 인터페이스용 테이블';
COMMENT ON COLUMN BAT_UP_IMDEPT.INS_TIME IS '생성일시';
COMMENT ON COLUMN BAT_UP_IMDEPT.DEPT_CODE IS '부서코드';
COMMENT ON COLUMN BAT_UP_IMDEPT.DEPT_NAME IS '부서명';
COMMENT ON COLUMN BAT_UP_IMDEPT.UP_DEPT_CD IS '부서명';
COMMENT ON COLUMN BAT_UP_IMDEPT.PRNT_CD IS '상위부서코드';
COMMENT ON COLUMN BAT_UP_IMDEPT.MANAGER_ID IS '담당자ID';
COMMENT ON COLUMN BAT_UP_IMDEPT.IF_FLAG IS '처리여부';
